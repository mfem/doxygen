21void VectorMassIntegrator::AssemblePA(const FiniteElementSpace &fes)
22{
23 Mesh *mesh = fes.GetMesh();
24 const FiniteElement &el = *fes.GetTypicalFE();
25 MFEM_VERIFY(el.GetMapType() == FiniteElement::VALUE,
26 "Only value map type supported");
27 ElementTransformation &Trans = *mesh->GetTypicalElementTransformation();
28 const auto *ir = IntRule ? IntRule : &MassIntegrator::GetRule(el, el, Trans);
29
30 if (DeviceCanUseCeed())
31 {
32 delete ceedOp;
33 const bool mixed =
34 mesh->GetNumGeometries(mesh->Dimension()) > 1 || fes.IsVariableOrder();
35 if (mixed) { ceedOp = new ceed::MixedPAMassIntegrator(*this, fes, Q); }
36 else { ceedOp = new ceed::PAMassIntegrator(fes, *ir, Q); }
37 return;
38 }
39
40 // If vdim is not set, set it to the space dimension
41 vdim = (vdim == -1) ? Trans.GetSpaceDim() : vdim;
42 MFEM_VERIFY(vdim == fes.GetVDim(), "vdim != fes.GetVDim()");
43 MFEM_VERIFY(vdim == mesh->Dimension(), "vdim != dim");
44
45 const MemoryType mt = pa_mt == MemoryType::DEFAULT
46 ? Device::GetDeviceMemoryType()
47 : pa_mt;
48
49 ne = mesh->GetNE();
50 dim = mesh->Dimension();
51 const int nq = ir->GetNPoints();
52 const int sdim = mesh->SpaceDimension();
53 geom = mesh->GetGeometricFactors(*ir, GeometricFactors::JACOBIANS, mt);
54 maps = &el.GetDofToQuad(*ir, DofToQuad::TENSOR);
55 dofs1D = maps->ndof;
56 quad1D = maps->nqpt;
57 const int q1d = quad1D;
58
59 if (!(dim == 2 || dim == 3)) { MFEM_ABORT("Dimension not supported."); }
60
61 QuadratureSpace qs(*mesh, *ir);
62 CoefficientVector coeff(qs);
63
64 if (Q)
65 {
66 coeff.Project(*Q);
67 }
68 else if (VQ)
69 {
70 coeff.Project(*VQ);
71 MFEM_VERIFY(VQ->GetVDim() == vdim, "VQ vdim vs. vdim error");
72 }
73 else if (MQ)
74 {
75 coeff.ProjectTranspose(*MQ);
76 MFEM_VERIFY(MQ->GetVDim() == vdim, "MQ dimension vs. vdim error");
77 MFEM_VERIFY(coeff.Size() == (vdim*vdim) * ne * nq, "MQ size error");
78 }
79 else { coeff.SetConstant(1.0); }
80
81 coeff_vdim = coeff.GetVDim();
82 const bool const_coeff = coeff_vdim == 1;
83 const bool vector_coeff = coeff_vdim == vdim;
84 const bool matrix_coeff = coeff_vdim == vdim * vdim;
85 MFEM_VERIFY(const_coeff + vector_coeff + matrix_coeff == 1, "");
86
87 pa_data.SetSize(coeff_vdim * nq * ne, mt);
88
89 const auto w_r = ir->GetWeights().Read();
90
91 if (dim == 2)
92 {
93 const auto W = Reshape(w_r, q1d, q1d);
94 const auto C = Reshape(coeff.Read(), coeff_vdim, q1d, q1d, ne);
95 const auto J = Reshape(geom->J.Read(), q1d, q1d, sdim, dim, ne);
96 auto D = Reshape(pa_data.Write(), q1d, q1d, coeff_vdim, ne);
97
98 mfem::forall_2D(ne, q1d, q1d, [=] MFEM_HOST_DEVICE(int e)
99 {
100 MFEM_FOREACH_THREAD(qy, y, q1d)
101 {
102 MFEM_FOREACH_THREAD(qx, x, q1d)
103 {
104 const real_t J11 = J(qx, qy, 0, 0, e), J12 = J(qx, qy, 1, 0, e);
105 const real_t J21 = J(qx, qy, 0, 1, e), J22 = J(qx, qy, 1, 1, e);
106 const real_t detJ = (J11 * J22) - (J21 * J12);
107 const real_t w_det = W(qx, qy) * detJ;
108 D(qx, qy, 0, e) = C(0, qx, qy, e) * w_det;
109 if (const_coeff) { continue; }
110 D(qx, qy, 1, e) = C(1, qx, qy, e) * w_det;
111 if (vector_coeff) { continue; }
112 assert(matrix_coeff);
113 D(qx, qy, 2, e) = C(2, qx, qy, e) * w_det;
114 D(qx, qy, 3, e) = C(3, qx, qy, e) * w_det;
115 }
116 }
117 });
118 }
119 else if (dim == 3)
120 {
121 const auto W = Reshape(w_r, q1d, q1d, q1d);
122 const auto C = Reshape(coeff.Read(), coeff_vdim, q1d, q1d, q1d, ne);
123 const auto J = Reshape(geom->J.Read(), q1d, q1d, q1d, sdim, dim, ne);
124 auto D = Reshape(pa_data.Write(), q1d, q1d, q1d, coeff_vdim, ne);
125
126 mfem::forall_3D(ne, q1d, q1d, q1d, [=] MFEM_HOST_DEVICE(int e)
127 {
128 MFEM_FOREACH_THREAD(qz, z, q1d)
129 {
130 MFEM_FOREACH_THREAD(qy, y, q1d)
131 {
132 MFEM_FOREACH_THREAD(qx, x, q1d)
133 {
134 const real_t J11 = J(qx, qy, qz, 0, 0, e),
135 J12 = J(qx, qy, qz, 0, 1, e),
136 J13 = J(qx, qy, qz, 0, 2, e);
137 const real_t J21 = J(qx, qy, qz, 1, 0, e),
138 J22 = J(qx, qy, qz, 1, 1, e),
139 J23 = J(qx, qy, qz, 1, 2, e);
140 const real_t J31 = J(qx, qy, qz, 2, 0, e),
141 J32 = J(qx, qy, qz, 2, 1, e),
142 J33 = J(qx, qy, qz, 2, 2, e);
143 const real_t detJ = J11 * (J22 * J33 - J32 * J23) -
144 J21 * (J12 * J33 - J32 * J13) +
145 J31 * (J12 * J23 - J22 * J13);
146 const real_t w_det = W(qx, qy, qz) * detJ;
147 D(qx, qy, qz, 0, e) = C(0, qx, qy, qz, e) * w_det;
148 if (const_coeff) { continue; }
149 D(qx, qy, qz, 1, e) = C(1, qx, qy, qz, e) * w_det;
150 D(qx, qy, qz, 2, e) = C(2, qx, qy, qz, e) * w_det;
151 if (vector_coeff) { continue; }
152 D(qx, qy, qz, 3, e) = C(3, qx, qy, qz, e) * w_det;
153 D(qx, qy, qz, 4, e) = C(4, qx, qy, qz, e) * w_det;
154 D(qx, qy, qz, 5, e) = C(5, qx, qy, qz, e) * w_det;
155 D(qx, qy, qz, 6, e) = C(6, qx, qy, qz, e) * w_det;
156 D(qx, qy, qz, 7, e) = C(7, qx, qy, qz, e) * w_det;
157 D(qx, qy, qz, 8, e) = C(8, qx, qy, qz, e) * w_det;
158 }
159 }
160 }
161 });
162 }
163 else
164 {
165 MFEM_ABORT("Unknown VectorMassIntegrator::AssemblePA kernel for"
166 << " dim:" << dim << ", vdim:" << vdim << ", sdim:" << sdim);
167 }
168}
