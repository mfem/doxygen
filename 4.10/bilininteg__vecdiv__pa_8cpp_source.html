19// PA Divergence Assemble 2D kernel
20static void PADivergenceSetup2D(const int Q1D,
21 const int NE,
22 const Array<real_t> &w,
23 const Vector &j,
24 const real_t COEFF,
25 Vector &op)
26{
27 const auto W = Reshape(w.Read(), Q1D, Q1D);
28 const auto J = Reshape(j.Read(), Q1D, Q1D, 2, 2, NE);
29 auto y = Reshape(op.Write(), Q1D, Q1D, 2, 2, NE);
30
31 mfem::forall_2D(NE, Q1D, Q1D, [=] MFEM_HOST_DEVICE (int e)
32 {
33 MFEM_FOREACH_THREAD_DIRECT(qx, x, Q1D)
34 {
35 MFEM_FOREACH_THREAD_DIRECT(qy, y, Q1D)
36 {
37 const real_t J11 = J(qx, qy, 0, 0, e);
38 const real_t J21 = J(qx, qy, 1, 0, e);
39 const real_t J12 = J(qx, qy, 0, 1, e);
40 const real_t J22 = J(qx, qy, 1, 1, e);
41 const real_t cw = W(qx, qy) * COEFF;
42 y(qx, qy, 0, 0, e) = cw * J22;
43 y(qx, qy, 0, 1, e) = cw * -J12;
44 y(qx, qy, 1, 0, e) = cw * -J21;
45 y(qx, qy, 1, 1, e) = cw * J11;
46 }
47 }
48 });
49}

51// PA Divergence Assemble 3D kernel
52static void PADivergenceSetup3D(const int Q1D,
53 const int NE,
54 const Array<real_t> &w,
55 const Vector &j,
56 const real_t COEFF,
57 Vector &op)
58{
59 const auto W = Reshape(w.Read(), Q1D, Q1D, Q1D);
60 const auto J = Reshape(j.Read(), Q1D, Q1D, Q1D, 3, 3, NE);
61 auto y = Reshape(op.Write(), Q1D, Q1D, Q1D, 3, 3, NE);
62
63 mfem::forall_3D(NE, Q1D, Q1D, Q1D, [=] MFEM_HOST_DEVICE (int e)
64 {
65 MFEM_FOREACH_THREAD_DIRECT(qx, x, Q1D)
66 {
67 MFEM_FOREACH_THREAD_DIRECT(qy, y, Q1D)
68 {
69 MFEM_FOREACH_THREAD_DIRECT(qz, z, Q1D)
70 {
71 const real_t J11 = J(qx, qy, qz, 0, 0, e);
72 const real_t J21 = J(qx, qy, qz, 1, 0, e);
73 const real_t J31 = J(qx, qy, qz, 2, 0, e);
74 const real_t J12 = J(qx, qy, qz, 0, 1, e);
75 const real_t J22 = J(qx, qy, qz, 1, 1, e);
76 const real_t J32 = J(qx, qy, qz, 2, 1, e);
77 const real_t J13 = J(qx, qy, qz, 0, 2, e);
78 const real_t J23 = J(qx, qy, qz, 1, 2, e);
79 const real_t J33 = J(qx, qy, qz, 2, 2, e);
80 const real_t cw = W(qx, qy, qz) * COEFF;
81 // adj(J)
82 const real_t A11 = (J22 * J33) - (J23 * J32);
83 const real_t A12 = (J32 * J13) - (J12 * J33);
84 const real_t A13 = (J12 * J23) - (J22 * J13);
85 const real_t A21 = (J31 * J23) - (J21 * J33);
86 const real_t A22 = (J11 * J33) - (J13 * J31);
87 const real_t A23 = (J21 * J13) - (J11 * J23);
88 const real_t A31 = (J21 * J32) - (J31 * J22);
89 const real_t A32 = (J31 * J12) - (J11 * J32);
90 const real_t A33 = (J11 * J22) - (J12 * J21);
91 // Store wq * Q * adj(J)
92 y(qx, qy, qz, 0, 0, e) = cw * A11;
93 y(qx, qy, qz, 0, 1, e) = cw * A12;
94 y(qx, qy, qz, 0, 2, e) = cw * A13;
95 y(qx, qy, qz, 1, 0, e) = cw * A21;
96 y(qx, qy, qz, 1, 1, e) = cw * A22;
97 y(qx, qy, qz, 1, 2, e) = cw * A23;
98 y(qx, qy, qz, 2, 0, e) = cw * A31;
99 y(qx, qy, qz, 2, 1, e) = cw * A32;
100 y(qx, qy, qz, 2, 2, e) = cw * A33;
101 }
102 }
103 }
104 });
105}

107static void PADivergenceSetup(const int dim,
108 const int Q1D,
109 const int NE,
110 const Array<real_t> &W,
111 const Vector &J,
112 const real_t COEFF,
113 Vector &op)
114{
115 if (dim == 1) { MFEM_ABORT("dim==1 not supported in PADivergenceSetup"); }
116 else if (dim == 2)
117 {
118 PADivergenceSetup2D(Q1D, NE, W, J, COEFF, op);
119 }
120 else if (dim == 3)
121 {
122 PADivergenceSetup3D(Q1D, NE, W, J, COEFF, op);
123 }
124}

126void VectorDivergenceIntegrator::AssemblePA(const FiniteElementSpace &trial_fes,
127 const FiniteElementSpace &test_fes)
128{
129 // Assumes tensor-product elements ordered by nodes
130 MFEM_ASSERT(trial_fes.GetOrdering() == Ordering::byNODES,
131 "PA Only supports Ordering::byNODES!");
132 auto *mesh = trial_fes.GetMesh();
133 const auto &trial_fe = *trial_fes.GetTypicalFE();
134 const auto &test_fe = *test_fes.GetTypicalFE();
135 const auto *ir = IntRule ? IntRule :
136 &GetRule(trial_fe, test_fe,
137 *mesh->GetTypicalElementTransformation());
138 const int dims = trial_fe.GetDim();
139 nq = ir->GetNPoints();
140 dim = mesh->Dimension();
141 ne = trial_fes.GetNE();
142 geom = mesh->GetGeometricFactors(*ir, GeometricFactors::JACOBIANS);
143 trial_maps = &trial_fe.GetDofToQuad(*ir, DofToQuad::TENSOR);
144 trial_dofs1D = trial_maps->ndof;
145 quad1D = trial_maps->nqpt;
146 test_maps = &test_fe.GetDofToQuad(*ir, DofToQuad::TENSOR);
147 test_dofs1D = test_maps->ndof;
148 MFEM_ASSERT(quad1D == test_maps->nqpt,
149 "PA requires test and trial space to have same number of "
150 "quadrature points!");
151 pa_data.SetSize(nq * dims * dims * ne, Device::GetMemoryType());
152
153 real_t coeff = 1.0;
154 if (Q)
155 {
156 auto *cQ = dynamic_cast<ConstantCoefficient *>(Q);
157 MFEM_VERIFY(cQ, "only ConstantCoefficient is supported!");
158 coeff = cQ->constant;
159 }
160 PADivergenceSetup(dim, quad1D, ne, ir->GetWeights(), geom->J, coeff, pa_data);
161}
