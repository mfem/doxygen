22void ParBilinearForm::pAllocMat()
23{
24 int nbr_size = pfes->GetFaceNbrVSize();
25
26 if (precompute_sparsity == 0 || fes->GetVDim() > 1)
27 {
28 if (keep_nbr_block)
29 {
30 mat = new SparseMatrix(height + nbr_size, width + nbr_size);
31 }
32 else
33 {
34 mat = new SparseMatrix(height, width + nbr_size);
35 }
36 return;
37 }
38
39 // the sparsity pattern is defined from the map: face->element->dof
40 const Table &lelem_ldof = fes->GetElementToDofTable(); // <-- dofs
41 const Table &nelem_ndof = pfes->face_nbr_element_dof; // <-- vdofs
42 Table elem_dof; // element + nbr-element <---> dof
43 if (nbr_size > 0)
44 {
45 // merge lelem_ldof and nelem_ndof into elem_dof
46 int s1 = lelem_ldof.Size(), s2 = nelem_ndof.Size();
47 const int *I1 = lelem_ldof.GetI(), *J1 = lelem_ldof.GetJ();
48 const int *I2 = nelem_ndof.GetI(), *J2 = nelem_ndof.GetJ();
49 const int nnz1 = I1[s1], nnz2 = I2[s2];
50
51 elem_dof.SetDims(s1 + s2, nnz1 + nnz2);
52
53 int *I = elem_dof.GetI(), *J = elem_dof.GetJ();
54 for (int i = 0; i <= s1; i++)
55 {
56 I[i] = I1[i];
57 }
58 for (int j = 0; j < nnz1; j++)
59 {
60 J[j] = J1[j];
61 }
62 for (int i = 0; i <= s2; i++)
63 {
64 I[s1+i] = I2[i] + nnz1;
65 }
66 for (int j = 0; j < nnz2; j++)
67 {
68 J[nnz1+j] = J2[j] + height;
69 }
70 }
71 // dof_elem x elem_face x face_elem x elem_dof (keep_nbr_block = true)
72 // ldof_lelem x lelem_face x face_elem x elem_dof (keep_nbr_block = false)
73 Table dof_dof;
74 {
75 Table face_dof; // face_elem x elem_dof
76 {
77 Table *face_elem = pfes->GetParMesh()->GetFaceToAllElementTable();
78 if (nbr_size > 0)
79 {
80 mfem::Mult(*face_elem, elem_dof, face_dof);
81 }
82 else
83 {
84 mfem::Mult(*face_elem, lelem_ldof, face_dof);
85 }
86 delete face_elem;
87 if (nbr_size > 0)
88 {
89 elem_dof.Clear();
90 }
91 }
92
93 if (keep_nbr_block)
94 {
95 Table dof_face;
96 Transpose(face_dof, dof_face, height + nbr_size);
97 mfem::Mult(dof_face, face_dof, dof_dof);
98 }
99 else
100 {
101 Table ldof_face;
102 {
103 Table face_ldof;
104 Table *face_lelem = fes->GetMesh()->GetFaceToElementTable();
105 mfem::Mult(*face_lelem, lelem_ldof, face_ldof);
106 delete face_lelem;
107 Transpose(face_ldof, ldof_face, height);
108 }
109 mfem::Mult(ldof_face, face_dof, dof_dof);
110 }
111 }
112
113 int *I = dof_dof.GetI();
114 int *J = dof_dof.GetJ();
115 int nrows = dof_dof.Size();
116 real_t *data = Memory<real_t>(I[nrows]);
117
118 mat = new SparseMatrix(I, J, data, nrows, height + nbr_size);
119 *mat = 0.0;
120
121 dof_dof.LoseData();
122}

124void ParBilinearForm::ParallelRAP(SparseMatrix &loc_A, OperatorHandle &A,
125 bool steal_loc_A)
126{
127 ParFiniteElementSpace &pfespace = *ParFESpace();
128
129 // Create a block diagonal parallel matrix
130 OperatorHandle A_diag(Operator::Hypre_ParCSR);
131 A_diag.MakeSquareBlockDiag(pfespace.GetComm(),
132 pfespace.GlobalVSize(),
133 pfespace.GetDofOffsets(),
134 &loc_A);
135
136 // Parallel matrix assembly using P^t A P (if needed)
137 if (IsIdentityProlongation(pfespace.GetProlongationMatrix()))
138 {
139 A_diag.SetOperatorOwner(false);
140 A.Reset(A_diag.As<HypreParMatrix>());
141 if (steal_loc_A)
142 {
143 HypreStealOwnership(*A.As<HypreParMatrix>(), loc_A);
144 }
145 }
146 else
147 {
148 OperatorHandle P(Operator::Hypre_ParCSR);
149 P.ConvertFrom(pfespace.Dof_TrueDof_Matrix());
150 A.MakePtAP(A_diag, P);
151 }
152}

163void ParBilinearForm::ParallelAssemble(OperatorHandle &A, SparseMatrix *A_local)
164{
165 A.Clear();
166
167 if (A_local == NULL) { return; }
168 MFEM_VERIFY(A_local->Finalized(), "the local matrix must be finalized");
169
170 OperatorHandle dA(A.Type()), Ph(A.Type()), hdA;
171
172 if (interior_face_integs.Size() == 0)
173 {
174 // construct a parallel block-diagonal matrix 'A' based on 'a'
175 dA.MakeSquareBlockDiag(pfes->GetComm(), pfes->GlobalVSize(),
176 pfes->GetDofOffsets(), A_local);
177 }
178 else
179 {
180 // handle the case when 'a' contains off-diagonal
181 int lvsize = pfes->GetVSize();
182 const HYPRE_BigInt *face_nbr_glob_ldof = pfes->GetFaceNbrGlobalDofMap();
183 HYPRE_BigInt ldof_offset = pfes->GetMyDofOffset();
184
185 Array<HYPRE_BigInt> glob_J(A_local->NumNonZeroElems());
186 int *J = A_local->GetJ();
187 for (int i = 0; i < glob_J.Size(); i++)
188 {
189 if (J[i] < lvsize)
190 {
191 glob_J[i] = J[i] + ldof_offset;
192 }
193 else
194 {
195 glob_J[i] = face_nbr_glob_ldof[J[i] - lvsize];
196 }
197 }
198
199 // TODO - construct dA directly in the A format
200 hdA.Reset(
201 new HypreParMatrix(pfes->GetComm(), lvsize, pfes->GlobalVSize(),
202 pfes->GlobalVSize(), A_local->GetI(), glob_J,
203 A_local->GetData(), pfes->GetDofOffsets(),
204 pfes->GetDofOffsets()));
205 // - hdA owns the new HypreParMatrix
206 // - the above constructor copies all input arrays
207 glob_J.DeleteAll();
208 dA.ConvertFrom(hdA);
209 }
210
211 // TODO - assemble the Dof_TrueDof_Matrix directly in the required format?
212 Ph.ConvertFrom(pfes->Dof_TrueDof_Matrix());
213 // TODO: When Ph.Type() == Operator::ANY_TYPE we want to use the Operator
214 // returned by pfes->GetProlongationMatrix(), however that Operator is a
215 // const Operator, so we cannot store it in OperatorHandle. We need a const
216 // version of class OperatorHandle, e.g. ConstOperatorHandle.
217
218 A.MakePtAP(dA, Ph);
219}

229void ParBilinearForm::AssembleSharedFaces(int skip_zeros)
230{
231 ParMesh *pmesh = pfes->GetParMesh();
232 FaceElementTransformations *T;
233 Array<int> vdofs1, vdofs2, vdofs_all;
234 DenseMatrix elemmat;
235
236 int nfaces = pmesh->GetNSharedFaces();
237 for (int i = 0; i < nfaces; i++)
238 {
239 T = pmesh->GetSharedFaceTransformations(i);
240 int Elem2NbrNo = T->Elem2No - pmesh->GetNE();
241 pfes->GetElementVDofs(T->Elem1No, vdofs1);
242 pfes->GetFaceNbrElementVDofs(Elem2NbrNo, vdofs2);
243 vdofs1.Copy(vdofs_all);
244 for (int j = 0; j < vdofs2.Size(); j++)
245 {
246 if (vdofs2[j] >= 0)
247 {
248 vdofs2[j] += height;
249 }
250 else
251 {
252 vdofs2[j] -= height;
253 }
254 }
255 vdofs_all.Append(vdofs2);
256 for (int k = 0; k < interior_face_integs.Size(); k++)
257 {
258 interior_face_integs[k]->
259 AssembleFaceMatrix(*pfes->GetFE(T->Elem1No),
260 *pfes->GetFaceNbrFE(Elem2NbrNo),
261 *T, elemmat);
262 if (keep_nbr_block)
263 {
264 mat->AddSubMatrix(vdofs_all, vdofs_all, elemmat, skip_zeros);
265 }
266 else
267 {
268 mat->AddSubMatrix(vdofs1, vdofs_all, elemmat, skip_zeros);
269 }
270 }
271 }

293void ParBilinearForm::AssembleDiagonal(Vector &diag) const
294{
295 MFEM_ASSERT(diag.Size() == fes->GetTrueVSize(),
296 "Vector for holding diagonal has wrong size!");
297 const Operator *P = fes->GetProlongationMatrix();
298 if (!ext)
299 {
300 MFEM_ASSERT(p_mat.Ptr(), "the ParBilinearForm is not assembled!");
301 p_mat->AssembleDiagonal(diag); // TODO: add support for PETSc matrices
302 return;
303 }
304 // Here, we have extension, ext.
305 if (IsIdentityProlongation(P))
306 {
307 ext->AssembleDiagonal(diag);
308 return;
309 }
310 // Here, we have extension, ext, and parallel/conforming prolongation, P.
311 Vector local_diag(P->Height());
312 ext->AssembleDiagonal(local_diag);
313 if (fes->Conforming())
314 {
315 P->MultTranspose(local_diag, diag);
316 return;
317 }
318 // For an AMR mesh, a convergent diagonal is assembled with |P^T| d_l,
319 // where |P^T| has the entry-wise absolute values of the conforming
320 // prolongation transpose operator.
321 const HypreParMatrix *HP = dynamic_cast<const HypreParMatrix*>(P);
322 if (HP)
323 {
324 HP->AbsMultTranspose(1.0, local_diag, 0.0, diag);
325 }
326 else
327 {
328 MFEM_ABORT("unsupported prolongation matrix type.");
329 }

385void ParBilinearForm::TrueAddMult(const Vector &x, Vector &y, const real_t a)
386const
387{
388 const Operator *P = pfes->GetProlongationMatrix();
389 Xaux.SetSize(P->Height());
390 Yaux.SetSize(P->Height());
391 Ytmp.SetSize(P->Width());
392
393 P->Mult(x, Xaux);
394 if (ext)
395 {
396 ext->Mult(Xaux, Yaux);
397 }
398 else
399 {
400 MFEM_VERIFY(interior_face_integs.Size() == 0,
401 "the case of interior face integrators is not"
402 " implemented");
403 mat->Mult(Xaux, Yaux);
404 }
405 P->MultTranspose(Yaux, Ytmp);
406 y.Add(a, Ytmp);
407}

410 const ParGridFunction &y) const
411{
412 MFEM_ASSERT(mat != NULL, "local matrix must be assembled");
413
414 real_t loc = InnerProduct(x, y);
415 real_t glob = 0.;
416
417 MPI_Allreduce(&loc, &glob, 1, MPITypeMap<real_t>::mpi_type, MPI_SUM,
418 pfes->GetComm());
419
420 return glob;

442{

454

456
457 delete Ax;
458
459 return res;

475void ParBilinearForm::FormLinearSystem(
476 const Array<int> &ess_tdof_list, Vector &x, Vector &b,
477 OperatorHandle &A, Vector &X, Vector &B, int copy_interior)
478{
479 const Operator &P = *pfes->GetProlongationMatrix();
480 const SparseMatrix &R = *pfes->GetRestrictionMatrix();
481
482 if (ext)
483 {
484 if (hybridization)
485 {
486 HypreParVector true_X(pfes), true_B(pfes);
487 P.MultTranspose(b, true_B);
488 R.Mult(x, true_X);
489
490 FormSystemMatrix(ess_tdof_list, A);
491
492 std::unique_ptr<ConstrainedOperator> A_constrained([&]()
493 {
494 Operator *op;
495 Operator::FormSystemOperator(ess_tdof_list, op);
496 return dynamic_cast<ConstrainedOperator*>(op);
497 }());
498 MFEM_ASSERT(A_constrained != nullptr, "");
499
500 A_constrained->EliminateRHS(true_X, true_B);
501 R.MultTranspose(true_B, b);
502 hybridization->ReduceRHS(true_B, B);
503 X.SetSize(B.Size());
504 X = 0.0;
505 }
506 else
507 {
508 ext->FormLinearSystem(ess_tdof_list, x, b, A, X, B, copy_interior);
509 }
510 return;
511 }
512
513 // Finish the matrix assembly and perform BC elimination, storing the
514 // eliminated part of the matrix.
515 FormSystemMatrix(ess_tdof_list, A);
516
517 // Transform the system and perform the elimination in B, based on the
518 // essential BC values from x. Restrict the BC part of x in X, and set the
519 // non-BC part to zero. Since there is no good initial guess for the Lagrange
520 // multipliers, set X = 0.0 for hybridization.
521 if (static_cond)
522 {
523 // Schur complement reduction to the exposed dofs
524 static_cond->ReduceSystem(x, b, X, B, copy_interior);
525 }
526 else if (hybridization)
527 {
528 // Reduction to the Lagrange multipliers system
529 HypreParVector true_X(pfes), true_B(pfes);
530 P.MultTranspose(b, true_B);
531 R.Mult(x, true_X);
532 ParallelEliminateTDofsInRHS(ess_tdof_list, true_X, true_B);
533 R.MultTranspose(true_B, b);
534 hybridization->ReduceRHS(true_B, B);
535 X.SetSize(B.Size());
536 X = 0.0;
537 }
538 else
539 {
540 // Variational restriction with P
541 X.SetSize(P.Width());
542 B.SetSize(X.Size());
543 P.MultTranspose(b, B);
544 R.Mult(x, X);
545 ParallelEliminateTDofsInRHS(ess_tdof_list, X, B);
546 if (!copy_interior) { X.SetSubVectorComplement(ess_tdof_list, 0.0); }
547 }
548}

550void ParBilinearForm::FormSystemMatrix(const Array<int> &ess_tdof_list,
551 OperatorHandle &A)
552{
553 if (ext)
554 {
555 if (hybridization)
556 {
557 const int remove_zeros = 0;
558 Finalize(remove_zeros);
559 hybridization->GetParallelMatrix(A);
560 }
561 else
562 {
563 ext->FormSystemMatrix(ess_tdof_list, A);
564 }
565 return;
566 }
567
568 // Finish the matrix assembly and perform BC elimination, storing the
569 // eliminated part of the matrix.
570 if (static_cond)
571 {
572 if (!static_cond->HasEliminatedBC())
573 {
574 static_cond->SetEssentialTrueDofs(ess_tdof_list);
575 static_cond->Finalize();
576 static_cond->EliminateReducedTrueDofs(Matrix::DIAG_ONE);
577 }
578 static_cond->GetParallelMatrix(A);
579 }
580 else
581 {
582 if (mat)
583 {
584 const int remove_zeros = 0;
585 Finalize(remove_zeros);
586 MFEM_VERIFY(p_mat.Ptr() == NULL && p_mat_e.Ptr() == NULL,
587 "The ParBilinearForm must be updated with Update() before "
588 "re-assembling the ParBilinearForm.");
589 ParallelAssemble(p_mat, mat);
590 delete mat;
591 mat = NULL;
592 delete mat_e;
593 mat_e = NULL;
594 ParallelEliminateTDofs(ess_tdof_list);
595 }
596 if (hybridization)
597 {
598 hybridization->GetParallelMatrix(A);
599 }
600 else
601 {
602 A = p_mat;
603 }
604 }

607void ParBilinearForm::RecoverFEMSolution(
608 const Vector &X, const Vector &b, Vector &x)
609{
610 if (ext && !hybridization)
611 {
612 ext->RecoverFEMSolution(X, b, x);
613 return;
614 }
615
616 const Operator &P = *pfes->GetProlongationMatrix();
617
618 if (static_cond)
619 {
620 // Private dofs back solve
621 static_cond->ComputeSolution(b, X, x);
622 }
623 else if (hybridization)
624 {
625 // Primal unknowns recovery
626 HypreParVector true_X(pfes), true_B(pfes);
627 P.MultTranspose(b, true_B);
628 const SparseMatrix &R = *pfes->GetRestrictionMatrix();
629 R.Mult(x, true_X); // get essential b.c. from x
630 hybridization->ComputeSolution(true_B, X, true_X);
631 x.SetSize(P.Height());
632 P.Mult(true_X, x);
633 }
634 else
635 {
636 // Apply conforming prolongation
637 x.SetSize(P.Height(), GetHypreMemoryType());
638 P.Mult(X, x);
639 }

671void ParMixedBilinearForm::AssembleSharedFaces(int skip_zeros)
672{
673 ParMesh *pmesh = trial_pfes->GetParMesh();
674 FaceElementTransformations *T;
675 Array<int> tr_vdofs1, tr_vdofs2, tr_vdofs_all;
676 Array<int> te_vdofs1, te_vdofs2, te_vdofs_all;
677 DenseMatrix elemmat;
678
679 int nfaces = pmesh->GetNSharedFaces();
680 for (int i = 0; i < nfaces; i++)
681 {
682 T = pmesh->GetSharedFaceTransformations(i);
683 int Elem2NbrNo = T->Elem2No - pmesh->GetNE();
684 trial_pfes->GetElementVDofs(T->Elem1No, tr_vdofs1);
685 test_pfes->GetElementVDofs(T->Elem1No, te_vdofs1);
686 trial_pfes->GetFaceNbrElementVDofs(Elem2NbrNo, tr_vdofs2);
687 test_pfes->GetFaceNbrElementVDofs(Elem2NbrNo, te_vdofs2);
688
689 tr_vdofs1.Copy(tr_vdofs_all);
690 for (int j = 0; j < tr_vdofs2.Size(); j++)
691 {
692 if (tr_vdofs2[j] >= 0)
693 {
694 tr_vdofs2[j] += width;
695 }
696 else
697 {
698 tr_vdofs2[j] -= width;
699 }
700 }
701 tr_vdofs_all.Append(tr_vdofs2);
702
703 if (keep_nbr_block)
704 {
705 te_vdofs1.Copy(te_vdofs_all);
706 for (int j = 0; j < te_vdofs2.Size(); j++)
707 {
708 if (te_vdofs2[j] >= 0)
709 {
710 te_vdofs2[j] += height;
711 }
712 else
713 {
714 te_vdofs2[j] -= height;
715 }
716 }
717 te_vdofs_all.Append(te_vdofs2);
718 }
719
720 for (int k = 0; k < interior_face_integs.Size(); k++)
721 {

780 A.Clear();

782 if (A_local == NULL) { return; }
783 MFEM_VERIFY(A_local->Finalized(), "the local matrix must be finalized");
784
785 OperatorHandle dA(A.Type()), hdA;

793 test_pfes->GetDofOffsets(),
794 trial_pfes->GetDofOffsets(),

796 }

799 // handle the case when 'a' contains off-diagonal
800 const int lvrows = test_pfes->GetVSize();

812 }
813 else
814 {
815 glob_J[i] = face_nbr_glob_lcol[J[i] - lvcols];
816 }
817 }

820 hdA.Reset(
821 new HypreParMatrix(trial_pfes->GetComm(), lvrows, test_pfes->GlobalVSize(),
822 trial_pfes->GlobalVSize(), A_local->GetI(), glob_J,

824 trial_pfes->GetDofOffsets()));

826 // - the above constructor copies all input arrays
827 glob_J.DeleteAll();

829 }

832
