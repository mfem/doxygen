20void MassIntegrator::AssembleEA_(Vector &ea_data,
21 const bool add)
22{
23 using internal::EAMassAssemble1D;
24 using internal::EAMassAssemble2D;
25 using internal::EAMassAssemble3D;
26
27 MFEM_VERIFY(maps->mode != DofToQuad::RAGGED_TENSOR,
28 "AssembleEA not implemented for ragged tensor bases");
29 const Array<real_t> &B = maps->B;
30 if (dim == 1)
31 {
32 auto kernel = EAMassAssemble1D<0,0>;
33 switch ((dofs1D << 4 ) | quad1D)
34 {
35 case 0x22: kernel = EAMassAssemble1D<2,2>; break;
36 case 0x33: kernel = EAMassAssemble1D<3,3>; break;
37 case 0x44: kernel = EAMassAssemble1D<4,4>; break;
38 case 0x55: kernel = EAMassAssemble1D<5,5>; break;
39 case 0x66: kernel = EAMassAssemble1D<6,6>; break;
40 case 0x77: kernel = EAMassAssemble1D<7,7>; break;
41 case 0x88: kernel = EAMassAssemble1D<8,8>; break;
42 case 0x99: kernel = EAMassAssemble1D<9,9>; break;
43 }
44 return kernel(ne,B,pa_data,ea_data,add,dofs1D,quad1D);
45 }
46 else if (dim == 2)
47 {
48 auto kernel = EAMassAssemble2D<0,0>;
49 switch ((dofs1D << 4 ) | quad1D)
50 {
51 case 0x22: kernel = EAMassAssemble2D<2,2>; break;
52 case 0x33: kernel = EAMassAssemble2D<3,3>; break;
53 case 0x44: kernel = EAMassAssemble2D<4,4>; break;
54 case 0x55: kernel = EAMassAssemble2D<5,5>; break;
55 case 0x66: kernel = EAMassAssemble2D<6,6>; break;
56 case 0x77: kernel = EAMassAssemble2D<7,7>; break;
57 case 0x88: kernel = EAMassAssemble2D<8,8>; break;
58 case 0x99: kernel = EAMassAssemble2D<9,9>; break;
59 }
60 return kernel(ne,B,pa_data,ea_data,add,dofs1D,quad1D);
61 }
62 else if (dim == 3)
63 {
64 auto kernel = EAMassAssemble3D<0,0>;
65 switch ((dofs1D << 4 ) | quad1D)
66 {
67 case 0x23: kernel = EAMassAssemble3D<2,3>; break;
68 case 0x34: kernel = EAMassAssemble3D<3,4>; break;
69 case 0x45: kernel = EAMassAssemble3D<4,5>; break;
70 case 0x56: kernel = EAMassAssemble3D<5,6>; break;
71 case 0x67: kernel = EAMassAssemble3D<6,7>; break;
72 case 0x78: kernel = EAMassAssemble3D<7,8>; break;
73 case 0x89: kernel = EAMassAssemble3D<8,9>; break;
74 }
75 return kernel(ne,B,pa_data,ea_data,add,dofs1D,quad1D);
76 }
77 MFEM_ABORT("Unknown kernel.");
78}
