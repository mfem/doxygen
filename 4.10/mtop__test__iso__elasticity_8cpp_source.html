29#include "mtop_solvers.hpp"
30
31using namespace std;
32using namespace mfem;
33
34constexpr auto MESH_TRI = MFEM_SOURCE_DIR "/miniapps/mtop/sq_2D_9_tri.mesh";
35constexpr auto MESH_QUAD = MFEM_SOURCE_DIR "/miniapps/mtop/sq_2D_9_quad.mesh";
36
37int main(int argc, char *argv[])
38{
39 // Initialize MPI and HYPRE.
40 Mpi::Init();
41 Hypre::Init();
42
43 // Parse command-line options.
44 const char *mesh_file = MESH_QUAD;
45 const char *device_config = "cpu";
46 int order = 2;
47 bool pa = false;
48 bool dfem = false;
49 bool mesh_tri = false;
50 bool mesh_quad = false;
51 int par_ref_levels = 1;
52 bool paraview = false;
53 bool visualization = true;
54
55 OptionsParser args(argc, argv);
56 args.AddOption(&mesh_file, "-m", "--mesh", "Mesh file to use.");
57 args.AddOption(&device_config, "-d", "--device",
58 "Device configuration string, see Device::Configure().");
59 args.AddOption(&order, "-o", "--order",
60 "Finite element order (polynomial degree).");
61 args.AddOption(&pa, "-pa", "--partial-assembly", "-no-pa",
62 "--no-partial-assembly", "Enable Partial Assembly.");
63 args.AddOption(&dfem, "-dfem", "--dFEM", "-no-dfem", "--no-dFEM",
64 "Enable or not dFEM.");
65 args.AddOption(&mesh_tri, "-tri", "--triangular", "-no-tri",
66 "--no-triangular", "Enable or not triangular mesh.");
67 args.AddOption(&mesh_quad, "-quad", "--quadrilateral", "-no-quad",
68 "--no-quadrilateral", "Enable or not quadrilateral mesh.");
69 args.AddOption(&par_ref_levels, "-prl", "--par-ref-levels",
70 "Number of times to refine the mesh uniformly in parallel.");
71 args.AddOption(&paraview, "-pv", "--paraview", "-no-pv", "--no-paraview",
72 "Enable or not ParaView visualization");
73 args.AddOption(&visualization, "-vis", "--visualization", "-no-vis",
74 "--no-visualization",
75 "Enable or disable GLVis visualization.");
76 args.ParseCheck();
77 MFEM_VERIFY(!(pa && dfem), "pa and dfem cannot be both set");
78
79 // Enable hardware devices such as GPUs, and programming models such as
80 // CUDA, OCCA, RAJA and OpenMP based on command line options.
81 Device device(device_config);
82 if (Mpi::Root()) { device.Print(); }
83
84 // Read the (serial) mesh from the given mesh file on all processors. We
85 // can handle triangular, quadrilateral, tetrahedral, hexahedral, surface
86 // and volume meshes with the same code.
87 Mesh mesh(mesh_tri ? MESH_TRI : mesh_quad ? MESH_QUAD : mesh_file, 1, 1);
88 const int dim = mesh.Dimension();
89
90 // Refine the serial mesh on all processors to increase the resolution. In
91 // this example we do 'ref_levels' of uniform refinement. We choose
92 // 'ref_levels' to be the largest number that gives a final mesh with no
93 // more than 1000 elements.
94 {
95 const int ref_levels =
96 (int)floor(log(1000. / mesh.GetNE()) / log(2.) / dim);
97 for (int l = 0; l < ref_levels; l++) { mesh.UniformRefinement(); }
98 }
99 if (Mpi::Root())
100 {
101 std::cout << "Number of elements: " << mesh.GetNE() << std::endl;
102 }
103
104 // Define a parallel mesh by a partitioning of the serial mesh. Refine
105 // this mesh further in parallel to increase the resolution. Once the
106 // parallel mesh is defined, the serial mesh can be deleted.
107 ParMesh pmesh(MPI_COMM_WORLD, mesh);
108 mesh.Clear();
109 for (int l = 0; l < par_ref_levels; l++) { pmesh.UniformRefinement(); }
110
111 // Create the solver
112 IsoLinElasticSolver elsolver(&pmesh, order, pa, dfem);
113 if (Mpi::Root())
114 {
115 std::cout << "Number of unknowns: "
116 << elsolver.GetSolutionVector().Size() << std::endl;
117 }
118
119 // set boundary conditions
120 elsolver.AddDispBC(2, -1, 0.0);
121 elsolver.AddDispBC(5, -1, 0.0);
122
123 // set material properties
124 ConstantCoefficient E(1.0), nu(0.2);
125 elsolver.SetMaterial(E, nu);
126
127 // set surface load
128 elsolver.AddSurfLoad(1, 0.0, 1.0);
129
130 // set convergence tolerances and max iterations
131 elsolver.SetLinearSolver(1e-6,1e-8,100);
132
133 // assemble the discrete system
134 elsolver.Assemble();
135
136 // solve the system
137 elsolver.FSolve();
138
139 // extract the solution
140 ParGridFunction &sol = elsolver.GetDisplacements();
141
142 if (paraview)
143 {
144 ParaViewDataCollection paraview_dc("isoel", &pmesh);
145 paraview_dc.SetPrefixPath("ParaView");
146 paraview_dc.SetLevelsOfDetail(order);
147 paraview_dc.SetDataFormat(VTKFormat::BINARY);
148 paraview_dc.SetHighOrderOutput(true);
149 paraview_dc.SetCycle(0);
150 paraview_dc.SetTime(0.0);
151 paraview_dc.RegisterField("disp", &sol);
152 paraview_dc.Save();
153 }
154
155 if (socketstream glvis; visualization &&
156 (glvis.open("localhost", 19916), glvis.is_open()))
157 {
158 glvis << "parallel " << Mpi::WorldSize() << " " << Mpi::WorldRank() << "\n";
159 glvis << "solution\n" << pmesh << sol << std::flush;
160 }
161
162 return EXIT_SUCCESS;
163}
