33void IntegerSet::Recreate(const int n, const int *p)
34{
35 int i, j;
36
37 SetSize(n);
38
39 for (i = 0; i < n; i++)
40 {
41 data[i] = p[i];
42 }
43
44 Sort();
45
46 for (j = 0, i = 1; i < n; i++)
47 if (data[i] != data[j])
