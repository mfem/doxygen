19void VectorDiffusionIntegrator::AssembleMF(const FiniteElementSpace &fes)
20{
21 // Assumes tensor-product elements
22 Mesh *mesh = fes.GetMesh();
23 const FiniteElement &el = *fes.GetTypicalFE();
24 const IntegrationRule *ir
25 = IntRule ? IntRule : &DiffusionIntegrator::GetRule(el, el);
26 if (DeviceCanUseCeed())
27 {
28 delete ceedOp;
29 MFEM_VERIFY(!VQ && !MQ,
30 "Only scalar coefficient supported for DiffusionIntegrator"
31 " with libCEED");
32 const bool mixed = mesh->GetNumGeometries(mesh->Dimension()) > 1 ||
33 fes.IsVariableOrder();
34 if (mixed)
35 {
36 ceedOp = new ceed::MixedMFDiffusionIntegrator(*this, fes, Q);
37 }
38 else
39 {
40 ceedOp = new ceed::MFDiffusionIntegrator(fes, *ir, Q);
41 }
42 return;
43 }
44 MFEM_ABORT("Error: VectorDiffusionIntegrator::AssembleMF only implemented"
45 " with libCEED");
46}
